# xformers static library for PPU
# This builds only the PPU-specific xformers components needed for memory-efficient attention

cmake_minimum_required(VERSION 3.18)

# Only build for PPU
if(NOT USE_PPU)
  return()
endif()

# Get the xformers root directory
# The xformers source is in xformers/xformers/ not just xformers/
set(XFORMERS_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/xformers)
set(XFORMERS_CSRC ${XFORMERS_ROOT}/csrc)

#################### xformers PPU static library ####################
# Only compile the following files:
# - attention/cuda/fmha/attention*.cu
# - autogen_ppu/impl/*.cu (PPU-generated CUTLASS kernels)

# Attention kernel sources
file(GLOB xformers_attention_cu "${XFORMERS_CSRC}/attention/cuda/fmha/attention*.cu")

# PPU-generated CUTLASS kernel sources (exact glob from original commit)
file(GLOB xformers_autogen_ppu_cu
    ${XFORMERS_CSRC}/attention/cuda/fmha/autogen_ppu/impl/*.cu
)

# Debug: show what we're compiling
message(STATUS "xformers PPU compiling:")
message(STATUS "  attention files: ${xformers_attention_cu}")
message(STATUS "  autogen_ppu files: ${xformers_autogen_ppu_cu}")

# Build xformers as a shared library for easy debugging and .so swapping
add_library(xformers_ppu SHARED
    ${xformers_attention_cu}
    ${xformers_autogen_ppu_cu}
)

# Set properties
set_target_properties(xformers_ppu PROPERTIES
    POSITION_INDEPENDENT_CODE ON
)

# Include directories
# Note: ${CMAKE_CURRENT_SOURCE_DIR} (third_party/xformers) must be included because
# attention.cu includes xformers headers as "xformers/csrc/..."
# Using PRIVATE because these are only needed for compiling xformers, not for consumers
target_include_directories(xformers_ppu PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${XFORMERS_ROOT}
    ${XFORMERS_CSRC}
    ${XFORMERS_CSRC}/attention
    ${XFORMERS_CSRC}/attention/cuda
    ${XFORMERS_CSRC}/attention/cuda/fmha
)

# Include torch headers (needed by xformers source files)
# Note: The autograd generated file is needed by torch/types.h
target_include_directories(xformers_ppu SYSTEM PRIVATE
    ${CMAKE_SOURCE_DIR}/aten/src
    ${CMAKE_BINARY_DIR}/aten/src
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_BINARY_DIR}
    ${CMAKE_SOURCE_DIR}/torch/csrc/api/include
    ${CMAKE_SOURCE_DIR}/torch/csrc/autograd/generated
)

# Include actlize (PPU-optimized)
target_include_directories(xformers_ppu SYSTEM PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/actlize/include
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/actlize/tools/util/include
)

# Add PPU-specific compile definitions
target_compile_definitions(xformers_ppu PRIVATE
    USE_PPU
    USE_AIU
    ACOMPUTE_VERSION=10000
)

# Add PPU-specific compile options for optimization
# These flags were previously applied via set_source_files_properties in caffe2/CMakeLists.txt
# but lost during refactoring when xformers became a separate library built via add_subdirectory
target_compile_options(xformers_ppu PRIVATE
    --use_fast_math
)

# Ensure autograd generated files are created before xformers compiles
# The variable_factories.h and other autograd files are generated by generate-torch-sources
add_dependencies(xformers_ppu generate-torch-sources)

# Export library for use in ATen
# Note: XFORMERS_ROOT is xformers/xformers, so we need the parent (xformers) for "xformers/csrc/..." includes
set(XFORMERS_PPU_LIBRARY xformers_ppu PARENT_SCOPE)
set(XFORMERS_PPU_INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR} ${XFORMERS_ROOT} ${XFORMERS_CSRC} PARENT_SCOPE)

# Install xformers_ppu to the export set (required for shared library)
install(TARGETS xformers_ppu
    EXPORT Caffe2Targets
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)
